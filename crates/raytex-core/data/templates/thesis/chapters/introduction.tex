\begin{definition}\label{def:suite}
  {{fr:Une suite réelle est une application $u \colon \mathbb{N} \to \mathbb{R}$.|en:A real sequence is a map $u \colon \mathbb{N} \to \mathbb{R}$.}}
\end{definition}

\begin{theorem}[Bolzano--Weierstrass]\label{thm:bw}
  {{fr:Toute suite réelle bornée admet une sous-suite convergente.|en:Every bounded real sequence has a convergent subsequence.}}
\end{theorem}
